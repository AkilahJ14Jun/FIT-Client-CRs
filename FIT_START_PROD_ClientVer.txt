@echo off
TITLE FIT - Fish Inventory Tracking
echo ========================================
echo Starting FIT Application at E:\fit
echo ========================================

REM 1. Start Database
echo [1/3] Starting Database (Docker)...
cd /d "E:\fit"
docker compose up -d

REM 2. Start Backend (serves frontend on port 3001)
echo [2/3] Starting Backend API...
cd server
start /b node dist/index.js

REM 3. Open Browser
echo [3/3] Opening Application...
timeout /t 5 > nul
start http://localhost:3001

echo.
echo ========================================
echo FIT Application is now running!
echo Frontend: http://localhost:3001
echo Backend API: http://localhost:3001/api
echo ========================================
echo Close this window to stop the backend (or use docker compose down)
pause
