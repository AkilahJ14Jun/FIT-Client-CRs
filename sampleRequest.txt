var request = require('request');
var fs = require('fs');
var options = {
  'method': 'POST',
  'url': 'https://{YOUR_HOST_NAME}/wapp/api/upload/media',
  'headers': {
    'X-API-KEY': '{API_KEY}'
  },
  formData: {
    'file': {
      'value': fs.createReadStream('/C:/csi/newfiles.pdf'),
      'options': {
        'filename': 'newfiles.pdf',
        'contentType': null
      }
    }
  }
};
request(options, function (error, response) {
  if (error) throw new Error(error);
  console.log(response.body);
});
